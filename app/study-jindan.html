/**
 * [다올105 학습관리센터] 학습 역량 진단 구글폼 자동 생성 및 HTML 웹앱 연동 스크립트
 * 사용법:
 * 1. 새 구글 스프레드시트 생성 -> [확장 프로그램] -> [Apps Script] 클릭
 * 2. 본 코드를 붙여넣고 저장(Ctrl+S) 후 상단에서 'setupLearningDiagnosticForm' 실행
 * 3. 우측 상단 [배포] -> [새 배포] -> 유형: [웹 앱], 액세스 권한: [모든 사용자]로 배포
 * 4. 발급된 웹 앱 URL을 HTML 분석 프로그램에 붙여넣으면 실시간 연동 완료!
 */

function setupLearningDiagnosticForm() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var form = FormApp.create('[학습관리센터] 학생 학습 역량 및 공부 체질 진단검사');
  form.setDescription('나에게 가장 잘 맞는 공부 방법과 과목별 전략을 찾기 위한 진단입니다. 솔직하게 답변해 주세요.');

  form.addTextItem().setTitle('학생 이름').setRequired(true);
  form.addListItem().setTitle('현재 학년').setChoiceValues(['중3', '고1', '고2', '고3']).setRequired(true);
  form.addTextItem().setTitle('학교명').setRequired(false);
  form.addTextItem().setTitle('고민 과목 또는 목표').setRequired(false);

  // Part 0. 기본 학습 성향 (4문항)
  form.addMultipleChoiceItem().setTitle('[성향1] 나에게 가장 가까운 공부 스타일은?')
      .setChoiceValues(['규범형', '탐구형', '행동형', '이상형']).setRequired(true);
  form.addScaleItem().setTitle('[성향2] 개념 이해 방식 (1: 세부 디테일/나무 중심 ~ 5: 전체 흐름/숲 중심)').setBounds(1, 5).setRequired(true);
  form.addScaleItem().setTitle('[성향3] 학습 동기 방향 (1: 내적 흥미/자기지향 ~ 5: 인정·소속감/사회지향)').setBounds(1, 5).setRequired(true);
  form.addScaleItem().setTitle('[성향4] 계획 실행 방식 (1: 기분·컨디션/감성적 ~ 5: 원칙·계획대로/이성적)').setBounds(1, 5).setRequired(true);

  // Part 1. 심리 리스크 (5문항)
  var p1 = [
    '1. 공부를 왜 해야 하는지 뚜렷한 목표나 이유를 모르겠다. (동기 부족)',
    '2. 시험을 볼 때 아는 문제도 틀릴까 봐 심하게 불안하거나 긴장한다. (시험 불안/완벽주의)',
    '3. 계획을 세워도 어차피 다 못 지킬 것 같아 시작조차 하기 싫을 때가 많다. (무기력/회피)',
    '4. 공부하다 모르는 게 나오면 내 능력이 부족해서라고 자책하며 금방 우울해진다. (낮은 효능감)',
    '5. 부모님이나 선생님의 간섭이 심하게 느껴져 반항심에 더 공부하기가 싫다. (외적 통제 거부)'
  ];
  p1.forEach(function(q) { form.addScaleItem().setTitle(q).setBounds(1, 5).setRequired(true); });

  // Part 2. 학습 전략 (10문항)
  var p2 = [
    '6. 책상에 앉기 전에 오늘 무슨 과목을, 언제까지 할지 구체적인 계획을 먼저 세운다. (시간관리)',
    '7. 공부할 때 스마트폰이나 게임 등 방해 요소를 내 스스로 통제하고 차단할 수 있다. (환경통제)',
    '8. 인강을 듣거나 해설지를 보고 이해한 것과 내가 직접 풀 수 있는 것을 명확히 구분한다. (메타인지)',
    '9. 틀린 문제가 나오면 정답만 확인하지 않고, 내가 왜 틀렸는지 원인을 반드시 분석한다. (메타인지)',
    '10. 공부를 시작하면 최소 40~50분 이상은 딴짓 없이 집중 상태를 유지한다. (집중력)',
    '11. 수업(인강)을 들을 때 무작정 다 적지 않고, 핵심 내용과 부연 설명을 구분하여 필기한다. (조직화)',
    '12. 새로운 단원을 공부할 때 세부 내용부터 보지 않고, 목차나 전체적인 흐름부터 파악한다. (총체적사고)',
    '13. 암기해야 할 내용이 많을 때, 나만의 표나 마인드맵 등으로 묶어서 정리해 본다. (조직화/기억)',
    '14. 모르는 문제가 생기면 바로 해설지를 보지 않고, 최소 10분 이상 스스로 고민해 본다. (과제집착력)',
    '15. 끝까지 고민해도 모르는 내용은 선생님, 친구 등을 적극적으로 활용해 반드시 알아낸다. (자원활용)'
  ];
  p2.forEach(function(q) { form.addScaleItem().setTitle(q).setBounds(1, 5).setRequired(true); });

  form.setDestination(FormApp.DestinationType.SPREADSHEET, ss.getId());
  Logger.log('구글폼 생성 완료! 설문 URL: ' + form.getPublishedUrl());
}

// HTML 웹앱에서 실시간으로 설문 응답을 불러오는 API 엔드포인트
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheets()[0];
  var rows = sheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', rows: rows }))
    .setMimeType(ContentService.MimeType.JSON);
}
💡 왜 이 방식이 가장 좋은가요? 구글 앱스크립트로 화면(UI)까지 만들면 보안 배너가 뜨고 A4 인쇄가 깨지지만, 위처럼 데이터 다리(`doGet`)만 앱스크립트로 두고 화면은 지금 보시는 GitHub HTML 웹앱을 쓰면 디자인·속도·구글폼 자동 연동을 모두 완벽하게 누릴 수 있습니다.

  • • Rule 3 (외골수 편차형 & 반항심 단독 경보): 전체 전략 평균이 3.2점 이상이어도 특정 핵심 역량이 1.5점 이하로 무너진 경우 핀셋 취약(외골수형) 경보를 띄우며, 5번(외적 통제 거부)이 4~5점이면 강압적 관리 금지 경고를 자동 삽입합니다.
